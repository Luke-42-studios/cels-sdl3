**3. Implementation in src/texture/sdl3_texture.c:**

Include headers: `<cels_sdl3.h>`, `"sdl3_internal.h"`, `<string.h>`.

Static globals:
```c
#define SDL3_MAX_RENDERERS 8

static struct {
    SDL_Renderer*      renderer;
    SDL3_TextureCache  cache;
} s_cache_table[SDL3_MAX_RENDERERS];
static int s_cache_count = 0;

static char s_asset_base_dir[SDL3_TEXTURE_PATH_MAX] = {0};
```

Implement `sdl3_set_asset_base_dir(const char* path)`: Copy path into `s_asset_base_dir` using `SDL_strlcpy`. Strip trailing `/` if present.

Implement `sdl3_texture_cache_for_renderer(SDL_Renderer* renderer)`:
- Linear scan `s_cache_table[0..s_cache_count-1]` for matching renderer pointer.
- If found, return `&s_cache_table[i].cache`.
- If not found and `s_cache_count < SDL3_MAX_RENDERERS`, create new entry: set renderer pointer, `memset` cache to 0, increment `s_cache_count`, return cache pointer.
- If table full, `SDL_Log` warning and return NULL.

Implement `sdl3_texture_cache_load(cache, renderer, path)`:
- Resolve full path: if `s_asset_base_dir[0]` is set, `SDL_snprintf(resolved, sizeof(resolved), "%s/%s", s_asset_base_dir, path)`. Otherwise `SDL_strlcpy(resolved, path, sizeof(resolved))`.
- Check if already cached: scan `cache->entries[0..count-1]` for matching `path` (compare against resolved path using `SDL_strcmp`). If found and `entry->texture` is non-NULL, increment `ref_count` and return `(uint32_t)(i + 1)` (1-based handle).
- Load new texture: call `IMG_LoadTexture(renderer, resolved)`. If returns NULL, return 0 (caller handles FAILED state).
- Set blend mode for alpha support: `SDL_SetTextureBlendMode(tex, SDL_BLENDMODE_BLEND)`.
- Find an empty slot (first entry where `texture == NULL`) or append at `count`. If `count >= SDL3_TEXTURE_CACHE_CAPACITY`, destroy the texture and `SDL_Log` cache full warning, return 0.
- Fill entry: `SDL_strlcpy(entry->path, resolved, SDL3_TEXTURE_PATH_MAX)`, set texture pointer, get dimensions via `SDL_GetTextureSize(tex, &entry->width, &entry->height)`, set `ref_count = 1`.
- Return `(uint32_t)(slot + 1)` (1-based handle).

Implement `sdl3_texture_cache_get(cache, handle)`:
- If `handle == 0 || handle > (uint32_t)cache->count`, return NULL.
- Return `cache->entries[handle - 1].texture`.

Implement `sdl3_texture_cache_get_size(cache, handle, w, h)`:
- Bounds check, then set `*w = entry->width`, `*h = entry->height`.

Implement `sdl3_texture_cache_release(cache, handle)`:
- Bounds check. Decrement `ref_count`. If `ref_count` reaches 0 and `texture` is non-NULL, call `SDL_DestroyTexture(entry->texture)`, set `entry->texture = NULL`, `entry->path[0] = '\0'`.

Implement `sdl3_texture_cache_invalidate(cache)`:
- Loop all entries: set `texture = NULL` (do NOT call SDL_DestroyTexture -- SDL_DestroyRenderer handles that), clear `path`, set `ref_count = 0`.
- Set `cache->count = 0`.

Implement `sdl3_texture_cache_remove_renderer(SDL_Renderer* renderer)`:
- Find the renderer in `s_cache_table`. Call `sdl3_texture_cache_invalidate` on its cache.
- Remove from table by shifting remaining entries down (or swap with last). Decrement `s_cache_count`.

**AVOID:**
- Do NOT call `SDL_DestroyTexture` in `sdl3_texture_cache_invalidate`. SDL_DestroyRenderer frees all associated textures automatically. Calling DestroyTexture on already-freed textures causes double-free crashes.
- Do NOT use a global cache shared across renderers. Each renderer MUST have its own cache because textures are renderer-affine.
- Do NOT store raw `SDL_Texture*` in the sprite component. Use the opaque handle (uint32_t index). Handles survive cache invalidation gracefully (stale handle -> NULL from cache_get).
- Do NOT use `IMG_Init()` or `IMG_Quit()`. SDL3_image 3.x auto-initializes and has no init/quit functions.

**2. Add SDL3_TextureLoadSystem in sdl3_module.c:**

Add a new system at OnLoad phase. This system runs each frame, scanning for sprites that need loading. Place it after the InputSystem definition but before the Window/Frame state systems (the phase annotation handles ordering -- OnLoad systems run in registration order).

```c
CEL_System(SDL3_TextureLoadSystem, .phase = OnLoad) {
    cel_query(SDL3_Sprite, SDL3_Renderer, SDL3_WindowComponent);
    cel_each(SDL3_Sprite, SDL3_Renderer, SDL3_WindowComponent) {
        if (SDL3_Sprite->state != SDL3_TEXTURE_NONE) continue;
        if (!SDL3_Sprite->texture_path) continue;
        if (!SDL3_Renderer->renderer) continue;

        SDL3_TextureCache* cache = sdl3_texture_cache_for_renderer(
            SDL3_Renderer->renderer);
        if (!cache) continue;

        cel_update(SDL3_Sprite) {
            SDL3_Sprite->state = SDL3_TEXTURE_LOADING;
            uint32_t handle = sdl3_texture_cache_load(
                cache, SDL3_Renderer->renderer, SDL3_Sprite->texture_path);

            if (handle != 0) {
                SDL3_Sprite->texture_handle = handle;
                SDL3_Sprite->state = SDL3_TEXTURE_READY;
            } else {
                SDL3_Sprite->state = SDL3_TEXTURE_FAILED;
                SDL_Log("SDL3: Failed to load texture '%s': %s",
                        SDL3_Sprite->texture_path, SDL_GetError());
            }
        }
    }
}
```

**Key detail:** The system queries `SDL3_Sprite, SDL3_Renderer, SDL3_WindowComponent` on the same entity. This means sprites MUST live on window entities (same entity that has the renderer). This is the simplest approach per the RESEARCH open question #1 recommendation.

**3. Add SDL3_SpriteRenderSystem in sdl3_module.c:**

Add a new system at OnRender phase. This draws all READY sprites. Place the definition after the TextureLoadSystem.

```c
CEL_System(SDL3_SpriteRenderSystem, .phase = OnRender) {
    cel_query(SDL3_Sprite, SDL3_Renderer, SDL3_WindowComponent);
    cel_each(SDL3_Sprite, SDL3_Renderer, SDL3_WindowComponent) {
        if (SDL3_WindowComponent->state == SDL3_WINDOW_MINIMIZED ||
            SDL3_WindowComponent->state == SDL3_WINDOW_CLOSING ||
            SDL3_WindowComponent->state == SDL3_WINDOW_CLOSED) continue;
        if (SDL3_Sprite->state != SDL3_TEXTURE_READY) continue;

        SDL3_TextureCache* cache = sdl3_texture_cache_for_renderer(
            SDL3_Renderer->renderer);
        if (!cache) continue;

        SDL_Texture* tex = sdl3_texture_cache_get(
            cache, SDL3_Sprite->texture_handle);
        if (!tex) continue;

        /* Apply per-sprite alpha modulation */
        if (SDL3_Sprite->alpha < 255) {
            SDL_SetTextureAlphaMod(tex, SDL3_Sprite->alpha);
        }

        /* Determine source rect: NULL = entire texture */
        const SDL_FRect* src = NULL;
        if (SDL3_Sprite->src_rect.w > 0 && SDL3_Sprite->src_rect.h > 0) {
            src = &SDL3_Sprite->src_rect;
        }

        /* Determine rotation center: NULL = center of dst_rect */
        const SDL_FPoint* center = NULL;
        if (SDL3_Sprite->center.x != 0 || SDL3_Sprite->center.y != 0) {
            center = &SDL3_Sprite->center;
        }

        SDL_RenderTextureRotated(
            SDL3_Renderer->renderer, tex,
            src, &SDL3_Sprite->dst_rect,
            SDL3_Sprite->angle, center,
            SDL3_Sprite->flip);

        /* Reset alpha mod to avoid state leakage to other sprites using same texture */
        if (SDL3_Sprite->alpha < 255) {
            SDL_SetTextureAlphaMod(tex, 255);
        }
    }
}
```

if (SDL3_WindowComponent->state == SDL3_WINDOW_CLOSING) {

    if (SDL3_Renderer->renderer) {
        /* Invalidate all cached textures for this renderer.
